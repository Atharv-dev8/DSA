class Solution {
    static int sumOfDigits(int n) {
       int iDigit = 0;
       int iSum =0;
        while(n!=0)
        {
            iDigit = n%10;
            iSum = iSum + iDigit;
            
            n = n/10;
        }
        return iSum;
        
    }
}
